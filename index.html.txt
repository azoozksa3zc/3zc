        // فتح وإغلاق القائمة الجانبية
        function toggleMenu() {
            document.getElementById('sidebar').classList.toggle('active');
            document.getElementById('overlay').classList.toggle('active');
        }

        // دالة تحديث حالة الشهادات والألوان تلقائياً
        function setBadge(elementId, status) {
            const badge = document.getElementById(elementId);
            if (status === "شغالة") {
                badge.innerText = "الشهادة: شغالة";
                badge.style.background = 'var(--success)';
                badge.style.color = '#000';
            } else {
                badge.innerText = "الشهادة: متوقفة";
                badge.style.background = 'var(--danger)';
                badge.style.color = '#fff';
            }
        }

        // تطبيق الإعدادات من ملف config.js فوراً عند فتح الموقع
        document.getElementById('main-discord').href = SITE_CONFIG.discordLink;
        document.getElementById('menu-discord').href = SITE_CONFIG.discordLink;

        document.getElementById('ios-hack').href = SITE_CONFIG.ios.hackLink;
        document.getElementById('ios-old').href = SITE_CONFIG.ios.oldLink;
        setBadge('ios-badge', SITE_CONFIG.ios.status);

        document.getElementById('android-hack').href = SITE_CONFIG.android.hackLink;
        document.getElementById('android-old').href = SITE_CONFIG.android.oldLink;
        setBadge('android-badge', SITE_CONFIG.android.status);

        document.getElementById('pc-hack').href = SITE_CONFIG.pc.hackLink;
        setBadge('pc-badge', SITE_CONFIG.pc.status);
